-- ============================================================
-- Orleans 3.8.0 MySQL Clustering Database Setup Script
-- Package: Microsoft.Orleans.Clustering.AdoNet 3.8.0
-- ADO.NET Invariant: MySql.Data.MySqlClient
--
-- Run order:
-- 1. This file (creates DB, tables, and seeds OrleansQuery)
-- ============================================================
CREATE DATABASE IF NOT EXISTS orleanscluster
    CHARACTER SET utf8mb4
    COLLATE utf8mb4_unicode_ci;
USE orleanscluster;
-- ============================================================
-- STEP 1: OrleansQuery table
-- Stores named SQL queries used internally by Orleans at runtime.
-- Must be populated with seed data (see INSERT section below).
-- ============================================================
CREATE TABLE IF NOT EXISTS OrleansQuery
(
    QueryKey VARCHAR(64) NOT NULL,
    QueryText VARCHAR(8000) NOT NULL,
    PRIMARY KEY (QueryKey)
);
-- ============================================================
-- STEP 2: OrleansMembershipVersionTable
-- One row per Orleans cluster (DeploymentId).
-- Tracks the cluster membership version for optimistic concurrency.
-- ============================================================
CREATE TABLE IF NOT EXISTS OrleansMembershipVersionTable
(
    DeploymentId VARCHAR(150) NOT NULL,
    Timestamp TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    Version INT NOT NULL DEFAULT 0,
    PRIMARY KEY (DeploymentId)
);
-- ============================================================
-- STEP 3: OrleansMembershipTable
-- One row per silo (cluster member).
-- Tracks address, status, and heartbeat of each silo.
-- ============================================================
CREATE TABLE IF NOT EXISTS OrleansMembershipTable
(
    DeploymentId VARCHAR(150) NOT NULL,
    Address VARCHAR(45) NOT NULL,
    Port INT NOT NULL,
    Generation INT NOT NULL,
    SiloName VARCHAR(150) NOT NULL,
    HostName VARCHAR(150) NOT NULL,
    Status INT NOT NULL,
    ProxyPort INT DEFAULT NULL,
    SuspectTimes VARCHAR(8000) DEFAULT NULL,
    StartTime DATETIME NOT NULL,
    IAmAliveTime DATETIME NOT NULL,
    PRIMARY KEY (DeploymentId, Address, Port, Generation),
    CONSTRAINT FK_MembershipTable_MembershipVersionTable
        FOREIGN KEY (DeploymentId)
        REFERENCES OrleansMembershipVersionTable (DeploymentId)
);
-- ============================================================
-- STEP 4: Seed OrleansQuery with required query definitions
-- These are the exact queries Orleans 3.x expects for MySQL.
-- Without these Orleans will fail to start with:
-- "Not all required queries found."
-- ============================================================
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('UpdateIAmAlivetimeKey',
'UPDATE OrleansMembershipTable SET
    IAmAliveTime = @IAmAliveTime
WHERE
    DeploymentId = @DeploymentId AND @DeploymentId IS NOT NULL
    AND Address = @Address AND @Address IS NOT NULL
    AND Port = @Port AND @Port IS NOT NULL
    AND Generation = @Generation AND @Generation IS NOT NULL;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('InsertMembershipVersionKey',
'INSERT INTO OrleansMembershipVersionTable
    (DeploymentId)
SELECT @DeploymentId
WHERE NOT EXISTS
(
    SELECT 1 FROM OrleansMembershipVersionTable WHERE DeploymentId = @DeploymentId AND @DeploymentId IS NOT NULL
);
SELECT ROW_COUNT();')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('InsertMembershipKey',
'INSERT INTO OrleansMembershipTable
    (DeploymentId, Address, Port, Generation, SiloName, HostName, Status, ProxyPort, StartTime, IAmAliveTime)
SELECT @DeploymentId, @Address, @Port, @Generation, @SiloName, @HostName, @Status, @ProxyPort, @StartTime, @IAmAliveTime
WHERE NOT EXISTS
(
    SELECT 1 FROM OrleansMembershipTable WHERE DeploymentId = @DeploymentId AND Address = @Address AND Port = @Port AND Generation = @Generation
);
UPDATE OrleansMembershipVersionTable
SET
    Timestamp = UTC_TIMESTAMP(),
    Version = Version + 1
WHERE
    DeploymentId = @DeploymentId AND Version = @Version AND ROW_COUNT() > 0;
SELECT ROW_COUNT();')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('UpdateMembershipKey',
'UPDATE OrleansMembershipTable
SET
    Status = @Status,
    SuspectTimes = @SuspectTimes,
    IAmAliveTime = @IAmAliveTime
WHERE
    DeploymentId = @DeploymentId AND @DeploymentId IS NOT NULL
    AND Address = @Address AND @Address IS NOT NULL
    AND Port = @Port AND @Port IS NOT NULL
    AND Generation = @Generation AND @Generation IS NOT NULL;
UPDATE OrleansMembershipVersionTable
SET
    Timestamp = UTC_TIMESTAMP(),
    Version = Version + 1
WHERE
    DeploymentId = @DeploymentId AND Version = @Version AND ROW_COUNT() > 0;
SELECT ROW_COUNT();')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('MembershipReadRowKey',
'SELECT v.DeploymentId, m.Address, m.Port, m.Generation, m.SiloName, m.HostName, m.Status, m.ProxyPort, m.SuspectTimes, m.StartTime, m.IAmAliveTime, v.Version
FROM
    OrleansMembershipVersionTable v LEFT OUTER JOIN OrleansMembershipTable m ON v.DeploymentId = m.DeploymentId
WHERE
    v.DeploymentId = @DeploymentId AND m.Address = @Address AND m.Port = @Port AND m.Generation = @Generation;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('MembershipReadAllKey',
'SELECT v.DeploymentId, m.Address, m.Port, m.Generation, m.SiloName, m.HostName, m.Status, m.ProxyPort, m.SuspectTimes, m.StartTime, m.IAmAliveTime, v.Version
FROM
    OrleansMembershipVersionTable v LEFT OUTER JOIN OrleansMembershipTable m ON v.DeploymentId = m.DeploymentId
WHERE
    v.DeploymentId = @DeploymentId;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('DeleteMembershipTableEntriesKey',
'DELETE m FROM OrleansMembershipTable m
WHERE m.DeploymentId = @DeploymentId;
DELETE v FROM OrleansMembershipVersionTable v
WHERE v.DeploymentId = @DeploymentId;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('GatewaysQueryKey',
'SELECT m.Address, m.ProxyPort, m.Generation
FROM
    OrleansMembershipVersionTable v JOIN OrleansMembershipTable m ON v.DeploymentId = m.DeploymentId
WHERE
    m.Status = @Status AND m.ProxyPort > 0 AND v.DeploymentId = @DeploymentId;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
INSERT INTO OrleansQuery (QueryKey, QueryText) VALUES
('CleanupDefunctSiloEntriesKey',
'DELETE FROM OrleansMembershipTable
WHERE
    DeploymentId = @DeploymentId
    AND @DeploymentId IS NOT NULL
    AND IAmAliveTime < @IAmAliveTime
    AND Status != @Status;')
ON DUPLICATE KEY UPDATE QueryText = VALUES(QueryText);
